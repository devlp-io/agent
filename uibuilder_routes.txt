@app.get("/api/schema")
def _harness_api_schema():
    try:
        rd = newest_run()
        if rd is None:
            return {"schema": {}}
        p = pathlib.Path(rd) / "schema.json"
        if p.exists():
            import json as _j
            return {"schema": _j.loads(p.read_text())}
    except Exception as e:
        return {"error": str(e)}
    return {"schema": {}}


@app.get("/api/recorder")
def _harness_api_recorder(limit: int = 500):
    try:
        return {"summary": _recorder.summary(TARGET),

                "http_proofs": _recorder.http_proofs(TARGET),

@app.get("/api/logs")
def _harness_api_logs():
    try:
        rd = newest_run()
        if rd is None:
            return {"run_dir": None, "files": {}}
        out = {"run_dir": str(rd), "files": {}}
        for f in ("live.log", "live_commands.log", "live_findings.log"):
            p = pathlib.Path(rd) / f
            if p.exists():
                out["files"][f] = p.read_text(errors="ignore").splitlines()[-400:]
            else:
                out["files"][f] = []
        return out
    except Exception as e:
        return {"error": str(e), "files": {}}


@app.post("/api/shell")
async def _harness_api_shell(request: Request):
    try:
        body = await request.json()
        cmd = body.get("cmd", "")
        if not cmd:
            return {"error": "cmd required"}
        rd = newest_run()
        if rd is None:
            return {"error": "no run dir"}
        out = _ag._run_shell(cmd, timeout=None)
        return {"cmd": cmd, "output": str(out)}
    except Exception as e:
        return {"error": str(e)}


@app.post("/api/directive")
async def _harness_api_directive(request: Request):
    try:
        body = await request.json()
        kind = (body.get("kind") or "").upper()
        payload = body.get("payload") or ""
        if not kind:
            return {"error": "kind required"}
        rd = newest_run()
        if rd is None:
            return {"error": "no run dir"}
        _ag.RUN_DIR = rd
        _ag.STATE = _state_mod.load(rd)
        return {"result": _ag.dispatch(kind, payload)}
    except Exception as e:
        return {"error": str(e)}


@app.post("/api/scan")
def _harness_api_scan(attack: int = 0, resume: int = 0, dry: int = 0):
    try:
        import subprocess as _sub
        args = [__import__("sys").executable, str(AGENT_ROOT / "agent_v8.py")]
        if attack: args.append("--attack")
        if resume: args.append("--resume")
        if dry: args.append("--dry")
        args.append("assess " + TARGET)
        proc = _sub.Popen(args, cwd=str(AGENT_ROOT),
                          stdout=_sub.DEVNULL, stderr=_sub.DEVNULL)
        return {"spawned": True, "pid": proc.pid}
    except Exception as e:
        return {"error": str(e)}


@app.post("/api/attack")
def _harness_api_attack():
    return _harness_api_scan(attack=1)


@app.post("/api/resume")
def _harness_api_resume():
    return _harness_api_scan(resume=1)
