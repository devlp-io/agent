import agent_v8 as _ag       # MODULE, not a class

_lock = threading.Lock()

def newest_run():
    base = SCAN_ROOT / TARGET
    if not base.exists(): return None
    runs = [p for p in base.iterdir() if p.is_dir() and p.name[:8].isdigit()]
    return sorted(runs)[-1] if runs else None

def bind():
    rd = newest_run()
    if rd is None: return None
    _ag.RUN_DIR = rd
    try: _ag.STATE = _state_mod.load(rd)
    except Exception: _ag.STATE = {}
    return _ag

@app.get("/api/pocs")
def api_pocs():
    """Scan state.json for CVE ids and report saved PoCs for each."""
    rd = newest_run()
    out = []
    if rd is None:
        return {"pocs": []}
    try:
        st = _state_mod.load(str(rd)) or {}
        import re as _re
        seen = set()
        for f in st.get("findings", []) or []:
            blob = (f.get("title") or "") + " " + (f.get("evidence") or "")
            for m in _re.finditer(r"CVE-\d{4}-\d{4,7}", blob, _re.I):
                cve = m.group(0).upper()
                if cve in seen:
                    continue
                seen.add(cve)
                try:
                    meta = _poc_mod.lookup(cve)
                    out.append({"cve": cve, "ok": meta.get("ok"),
                                "url": meta.get("url"),
                                "source": meta.get("source"),
                                "stars": meta.get("stars"),
                                "local_path": meta.get("local_path"),
                                "readme": (meta.get("readme") or "")[:1000]})
                except Exception:
                    pass
    except Exception:
        pass
    return {"pocs": out}

@app.get("/api/archive")
def api_archive():
    """List evidence archive files for the newest run."""
    rd = newest_run()
    if rd is None:
        return {"counts": {"http": 0, "tool": 0}, "index": []}
    try:
        from evidence import archive as _arch
        s = _arch.summary(str(rd))
    except Exception:
        s = {"http": 0, "tool": 0}
    idx = []
    ip = pathlib.Path(rd) / "archive" / "index.jsonl"
    if ip.exists():
        try:
            for line in ip.read_text(errors="ignore").splitlines()[-200:]:
                import json as _j
                idx.append(_j.loads(line))
        except Exception:
            pass
    return {"counts": s, "index": idx[-100:]}
